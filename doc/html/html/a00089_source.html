60 # Load variadic API
61 # Notes:
62 # - Remember to manually apply default argument promotions when calling variadic functions;
63 # anyway it is not necessary with ReadData since variadic arguments are always
64 # pointers, that are not subject to default argument promotion. More details on
65 # https://en.cppreference.com/w/c/language/conversion#Default_argument_promotions
66 # - On some platforms, like Apple ARM64, "it is required to specify the argtypes attribute
67 # for the regular, non-variadic, function arguments" (see ctypes documentation); the
68 # other variadic arguments could be set at runtime after a call to SetReadDataFormat,
69 # but this would not be safe because ReadData can be called concurrently from multiple
70 # threads with different signatures, so arguments (always pointers in this case) must
71 # be placed without relying on ctypes automatic conversions with from_param methods.
72 # For more details, see https://stackoverflow.com/q/74630617/3287591
73 self.read_data = self.__get('ReadData', ct.c_uint64, ct.c_int, variadic=True)

75 def __api_errcheck(self, res: int, func, _: tuple) -> int:
76 # res can be positive on GetChildHandles and GetDeviceTree
77 if res < 0:
78 raise error.Error(self.last_error, res, func.__name__)

81 def __get(self, name: str, *args: type, **kwargs) -> Callable[..., int]:
82 min_version = kwargs.get('min_version')
83 if min_version is not None:
84 if not self.ver_at_least(min_version):
85 def fallback(*args, **kwargs):
86 raise NotImplementedError(f'{name} requires {self.name} >= {min_version}.')
87 return fallback
88 func = self.get(f'CAEN_FELib_{name}', kwargs.get('variadic', False))
89 func.argtypes = args
90 func.restype = ct.c_int
91 func.errcheck = self.__api_errcheck # type: ignore
92 return func

96 def get_lib_info(self, initial_size: int = 2**22) -> dict:
97 """
98 Binding of CAEN_FELib_GetLibInfo()
99
100 @sa info
101 @param[in] initial_size initial size to be allocated for the first iteration
102 @return JSON representation of the library info (a dictionary)
103 """
104 while True:
105 lib_info = ct.create_string_buffer(initial_size)
106 res = self.__get_lib_info(lib_info, initial_size)
107 if res < initial_size: # equal not fine, see docs
108 return loads(lib_info.value.decode('ascii'))
109 initial_size = res
