15class Manager(list[_lru_cache_wrapper]):
16 """
17 A simple list of functions returned by `@lru_cache` decorator.
18
19 To be used with the optional parameter @p cache_manager of
20 lru_cache_method(), that will store a reference to the cached
21 function inside this list. This is a typing-safe way to call
22 `cache_clear` and `cache_info` of the internal cached functions,
23 even if not exposed directly by the inner function returned by
24 lru_cache_method().
25 """

27 """Invoke `cache_clear` on all functions in the list"""

36def cached(cache_manager: Optional[Manager] = None, maxsize: int = 128, typed: bool = False):
37 """
38 LRU cache decorator that keeps a weak reference to self.
39
40 To be used as decorator on methods that are known to return always
41 the same value. This can improve the performances of some methods by
42 a factor > 1000.
43 This wrapper using weak references is required: functools.lru_cache
44 holds a reference to all arguments: using directly on the methods it
45 would hold a reference to self, introducing subtle memory leaks.
46
47 @sa https://stackoverflow.com/a/68052994/3287591
48 """
49
50 def wrapper(method):
51
52 @lru_cache(maxsize, typed)
53 def cached_method(self_ref: ReferenceType, *args, **kwargs):
54 self = self_ref()
55 assert self is not None # this function is always called by inner()
56 return method(self, *args, **kwargs)
57
58 @wraps(method)
59 def inner(self, *args, **kwargs):
60 return cached_method(ref(self), *args, **kwargs)
61
62 # Optionally store a reference to lru_cache decorated function to
63 # simplify cache management. See CacheManager documentation.
64 if cache_manager is not None:
65 cache_manager.append(cached_method)
66
67 return inner
68
69 return wrapper
