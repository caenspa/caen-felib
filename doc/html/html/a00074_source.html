72class DppFlags(IntFlag):
73 """
74 Flags on Dig1 DPP events
75 """
76 DEADTIME = 0x0000001 # Identifies the first event after a dead time.
77 TTROLLOVER = 0x0000002 # Identifies a trigger time stamp roll-over that occurred before this event.
78 TTRESET = 0x0000004 # Identifies a trigger time stamp reset forced from external signals in S-IN (GPI for Desktop).
79 EVTFAKE = 0x0000008 # Identifies a fake event (which does not correspond to any physical event).
80 MEMFULL = 0x0000010 # Reading memory full.
81 TRGLOST = 0x0000020 # Identifies the first event after a trigger lost.
82 NTRGLOST = 0x0000040 # Every N lost events this flag is high (see board documentation to set N).
83 OVERRNG = 0x0000080 # Energy overranged.
84 F1024TRG = 0x0000100 # 1024 triggers counted (every 1024 counted events this flag is high).
85 LOSTEVT = 0x0000200 # Identifies the first event after when one or more events is lost due to a memory board FULL. The memory can be FULL due to a write event.
86 INPUTSAT = 0x0000400 # Identifies an event saturating the input dynamics (clipping).
87 NTRGTOT = 0x0000800 # Every N total events this flag is high (see board documentation to set N).
88 OLDSORT = 0x0001000 # Identifies an event not sorted but sent for waveform.
89 EOR = 0x0002000 # Identifies a fake event occurring at the end of run.
90 FINETT = 0x0004000 # Identifies an event with fine time stamp.
91 PILEUP = 0x0008000 # Identifies a pile up event.
92 TIMEVALUE = 0x0010000 # Identifies a fake event occurring on a time stamp roll-over.
93 ENERGY_SKIM = 0x0020000 # Energy skimming.
94 SATREJ = 0x0040000 # Identifies an event occurred when detector was inhibited due to saturation.
95 PLLLOCKLOSS = 0x0080000 # Identifies a fake event sent to report a PLL lock loss.
96 OVERTEMP = 0x0100000 # Identifies a fake event sent to report an over temperature condition.
97 SHUTDOWN = 0x0200000 # Identifies a fake event sent to report an ADC shutdown.
98 MEMORYSORT = 0x0400000 # Identifies a fake event sent to report a memory overload during data sorting.
99 MCS = 0x0800000 # Identifies a MCS event
100 STOPCOND = 0x1000000 # Identifies the first event after a stop condition has been met.
