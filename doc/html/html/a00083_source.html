68@dataclass(**_utils.dataclass_slots)
69class Data:
70 """
71 Class representing data set by Node.set_read_data_format(). It holds
72 a `numpy.ndarray` in value allocated with shape specified in the
73 data format.
74 """
75
76 # Public members
77 name: str
78 type: DataType
79 dim: int = field(default=0, repr=False)
80 shape: list[int] = field(default_factory=list, repr=False)
81
82 # Private members
83 __value: np.ndarray = field(init=False, repr=False)
84 __iliffe_vector_proxy: Optional[np.ndarray] = field(default=None, repr=False)
85 __arg: Any = field(init=False, repr=False)
86
87 def __post_init__(self) -> None:
88 if self.dim > 2:
89 raise ValueError('dim cannot be larger than 2')
90 if self.dim != len(self.shape):
91 raise ValueError('shape length must match dim')
92 dtype = _DATA_TYPE_MAP.get(self.type)
93 if dtype is None:
94 raise ValueError('Invalid data type')
95 # Memory allocation
96 self.__value = np.empty(self.shape, dtype=dtype)
97 self.__arg = self.__generate_arg()
98
99 @property
100 def value(self) -> np.ndarray:
101 """Instance of `numpy.ndarray` that holds data"""
102 return self.__value
103
104 @property
105 def arg(self) -> Any:
106 """
107 ctypes pointer to Data.value that is used within Node.read_data
108 """
109 return self.__arg
110
111 def __generate_arg(self) -> Any:
112 value: np.ndarray
113 if self.dim < 2:
114 # NumPy 0D and 1D arrays can be used directly.
115 value = self.value
116 else:
117 # NumPy 2D arrays cannot be directly used because they are
118 # implemented as contiguous memory blocks instead of Iliffe
119 # vectors, used by CAEN_FELib.
120 # To overcome the problem we generate a proxy ndarray of
121 # pointers, that must be stored locally to prevent garbage
122 # collection of the original array.
123 ptr_gen = (v.ctypes.data for v in self.value)
124 self.__iliffe_vector_proxy = np.fromiter(ptr_gen, dtype=ct.c_void_p)
125 value = self.__iliffe_vector_proxy
126 # value.ctypes is equivalent to
127 # value.ctypes.data_as(ctypes.c_void_p), that is fine for us.
128 return value.ctypes

130 def __str__(self) -> str:
131 return self.name

383 def set_read_data_format(self, fmt: Sequence[Union[Mapping[str, Any]]]) -> tuple[Data, ...]:
384 """
385 Binding of CAEN_FELib_SetReadDataFormat()
386
387 In addition to what happens in C library, it also allocate data.
388 Size of fields with `dim > 0` must be specified by a `"shape"`
389 entry in the field description, that is a vector passed to the
390 `shape` argument of `np.empty` constructor. On fields with
391 `dim == 0` the shape can be omitted, and is set to `[]` by
392 default. Fields can be accessed on data attribute of this class,
393 that is a list of Data initialized with the field descriptions,
394 in the same order of @p format.
395
396 Example:
397 ```
398 nch = dig.par.numch.value
399 reclen = dig.par.recordlengths.value
400 format = [
401 {
402 'name': 'WAVEFORM',
403 'type': device.DataType.U16,
404 'dim': 2,
405 'shape': [nch, reclen],
406 },
407 ]
408 data = ep_node.set_read_data_format(format)
409 print(data[0])
410 ```
411
412 @param[in] fmt JSON representation of the format, in compliance with the endpoint "format" property (a list of dictionaries)
413 @return Tuple of Data with allocated buffers of specified dim and shape, to be passed as second argument of read_data()
414 @exception error.Error in case of error
415 """
416 lib.set_read_data_format(self.handle, _utils.to_bytes(dumps(fmt)))
417
418 # Important:
419 # Do not update lib.ReadData.argtypes with data.argtype because lib.ReadData
420 # is shared with all other endpoints and it would not be thread safe.
421 # More details on a comment on the lib._Lib constructor.
422 # Possible unsafe code could be something like:
423 # lib.ReadData.argtypes = [ct.c_uint64, ct.c_int] + [d.argtype for d in self.data]
424
425 # Allocate requested fields
426 return tuple(Data(**f) for f in fmt)

428 def read_data(self, timeout: int, data: Sequence[Data]) -> None:
429 """
430 Binding of CAEN_FELib_ReadData()
431
432 Unlike what happens in C library, variadic arguments are added
433 automatically according to what has been specified by
434 set_read_data_format(). Data can be retrieved using the data
435 attribute of this class.
436
437 Example:
438 ```
439 # Start acquisition
440 dig.cmd.armacquisition()
441 dig.cmd.swstartacquisition()
442
443 while True:
444 try:
445 ep_node.read_data(100, data)
446 except error.Error as ex:
447 if ex.code is error.ErrorCode.TIMEOUT:
448 continue
449 elif ex.code is error.ErrorCode.STOP:
450 break
451 else:
452 raise ex
453
454 # Do stuff with data
455 for field in data:
456 print(field.value)
457
458 dig.cmd.disarmacquisition()
459 ```
460
461 @param[in] timeout timeout of the function in milliseconds; if this value is -1 the function is blocking with infinite timeout
462 @param[out] data The object returned by set_read_data_format().
463 @exception error.Error in case of error
464 """
465 lib.read_data(self.handle, timeout, *(d.arg for d in data))
