32 def __load_lib(self) -> None:
33 """
34 Variadic functions are __cdecl even if declared as __stdcall.
35 This difference applies only to 32 bit applications, 64 bit
36 applications have its own calling convention.
37 """
38 self.__lib_variadic: ct.CDLL
39 if sys.platform == 'win32':
40 self.__lib: Union[ct.CDLL, ct.WinDLL]
41 else:
42 self.__lib: ct.CDLL
43
44 # Load library
45 try:
46 if sys.platform == 'win32':
47 self.__path = f'{self.name}.dll'
48 if self.__stdcall:
49 self.__lib = ct.windll.LoadLibrary(self.__path)
50 self.__lib_variadic = ct.cdll.LoadLibrary(self.__path)
51 else:
52 self.__lib = ct.cdll.LoadLibrary(self.__path)
53 self.__lib_variadic = self.__lib
54 else:
55 self.__path = f'lib{self.name}.so'
56 self.__lib = ct.cdll.LoadLibrary(self.__path)
57 self.__lib_variadic = self.__lib
58
59 except _LibNotFoundClass as ex:
60 raise RuntimeError(
61 f'Library {self.name} not found. This module requires '
62 'the latest version of the library to be installed on '
63 'your system. You may find the official installers at '
64 'https://www.caen.it/. Please install it and retry.'
65 ) from ex
